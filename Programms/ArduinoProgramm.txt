#include <Arduino.h>
#include <LiquidCrystal.h>
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

/// seting up what the inputs are
void setup() {
  pinMode(A0,INPUT);
  Serial.begin(9600);
  lcd.begin(16, 2);
 
}
/* hardest thing to do is converting the data
I measured the actual voltage of my Arduino and it is 4.6
So I adjusted it and now the resuslts I get are way better */
void loop() {
  int raw_input;
  raw_input = analogRead(A0);
  float voltage = raw_input * (4.6 / 1023.0);
  float temperature = (voltage - 0.5) / 0.01;
  Serial.println(temperature);
  delay(2500);
  lcd.clear();
  lcd.print("temperature");
  lcd.setCursor(0, 1);
  lcd.print(temperature);
  lcd.print(" C");
}
